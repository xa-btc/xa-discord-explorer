cmake_minimum_required(VERSION 3.24)
project(XADiscordExplorer VERSION 0.5.0 LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(sqlite STATIC third_party/sqlite3.c)
target_compile_definitions(sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_DQS=0)
add_library(member_core STATIC native/core.cpp native/session.cpp native/gateway.cpp native/chat.cpp native/audit.cpp)
target_include_directories(member_core PUBLIC native third_party)
target_link_libraries(member_core PUBLIC sqlite bcrypt crypt32 winhttp)
target_compile_definitions(member_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_executable(XADiscordExplorer WIN32 native/app.cpp native/app.rc)
target_link_libraries(XADiscordExplorer PRIVATE member_core comctl32 comdlg32 shell32 dwmapi uxtheme)
set(XA_DISCORD_EXPLORER_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}" CACHE PATH "Application output directory")
set_target_properties(XADiscordExplorer PROPERTIES OUTPUT_NAME "XA Discord Explorer"
  RUNTIME_OUTPUT_DIRECTORY "${XA_DISCORD_EXPLORER_OUTPUT_DIRECTORY}"
  RUNTIME_OUTPUT_DIRECTORY_RELEASE "${XA_DISCORD_EXPLORER_OUTPUT_DIRECTORY}"
  RUNTIME_OUTPUT_DIRECTORY_DEBUG "${CMAKE_BINARY_DIR}/Debug")
if(MSVC)
  target_compile_options(member_core PRIVATE /W4 /utf-8)
  target_compile_options(XADiscordExplorer PRIVATE /W4 /utf-8)
  target_link_options(XADiscordExplorer PRIVATE "/MANIFESTDEPENDENCY:type='win32' name='Microsoft.Windows.Common-Controls' version='6.0.0.0' processorArchitecture='*' publicKeyToken='6595b64144ccf1df' language='*'")
endif()
enable_testing()
add_executable(member_tests native/tests.cpp)
target_link_libraries(member_tests PRIVATE member_core)
if(MSVC)
  target_compile_options(member_tests PRIVATE /W4 /utf-8)
endif()
add_test(NAME snapshot_behaviour COMMAND member_tests)
add_executable(member_ui_tests native/ui_tests.cpp)
target_link_libraries(member_ui_tests PRIVATE member_core comctl32 comdlg32 shell32 dwmapi uxtheme)
if(MSVC)
  target_compile_options(member_ui_tests PRIVATE /W4 /utf-8)
endif()
add_test(NAME collection_progress_ui COMMAND member_ui_tests)

add_executable(chat_tests native/chat_tests.cpp)
target_link_libraries(chat_tests PRIVATE member_core)
if(MSVC)
  target_compile_options(chat_tests PRIVATE /W4 /utf-8)
endif()
add_test(NAME message_collection COMMAND chat_tests)

add_executable(dm_tests native/dm_tests.cpp)
target_link_libraries(dm_tests PRIVATE member_core)
if(MSVC)
  target_compile_options(dm_tests PRIVATE /W4 /utf-8)
endif()
add_test(NAME dm_collection COMMAND dm_tests)
